def renew_ips(self):
    """Renew IPs for all active proxies"""
    try:
        # Validate API keys
        if not self.ak.get('access_key') or not self.ak.get('secret_key'):
            messagebox.showerror("Lỗi", "Vui lòng nhập Access Key và Secret Key!")
            return
        
        # Check if there are any instances to renew
        if not hasattr(self, 'instances') or not self.instances:
            messagebox.showwarning("Cảnh báo", "Không có instance nào để làm mới IP!")
            return
        
        # Show loading dialog
        loading = LoadingDialog(self.root, "Đang làm mới IP...")
        
        try:
            # Initialize EC2 client
            ec2_client = boto3.client(
                'ec2',
                aws_access_key_id=self.ak['access_key'],
                aws_secret_access_key=self.ak['secret_key'],
                region_name=self.ak.get('region', 'us-east-1')
            )
            
            renewed_count = 0
            failed_count = 0
            errors = []
            
            # Process each instance
            for instance_id in self.instances:
                try:
                    # Get current instance information
                    response = ec2_client.describe_instances(InstanceIds=[instance_id])
                    
                    if not response['Reservations']:
                        errors.append(f"Instance {instance_id}: Không tìm thấy")
                        failed_count += 1
                        continue
                    
                    instance = response['Reservations'][0]['Instances'][0]
                    
                    # Check if instance is running
                    if instance['State']['Name'] != 'running':
                        errors.append(f"Instance {instance_id}: Không đang chạy (trạng thái: {instance['State']['Name']})")
                        failed_count += 1
                        continue
                    
                    # Get network interface ID
                    network_interface_id = None
                    if 'NetworkInterfaces' in instance and instance['NetworkInterfaces']:
                        network_interface_id = instance['NetworkInterfaces'][0]['NetworkInterfaceId']
                    
                    if not network_interface_id:
                        errors.append(f"Instance {instance_id}: Không tìm thấy network interface")
                        failed_count += 1
                        continue
                    
                    # Get current association ID for the Elastic IP
                    association_id = None
                    allocation_id = None
                    
                    if 'Association' in instance['NetworkInterfaces'][0]:
                        association_id = instance['NetworkInterfaces'][0]['Association'].get('AssociationId')
                        allocation_id = instance['NetworkInterfaces'][0]['Association'].get('AllocationId')
                    
                    # Disassociate current Elastic IP if exists
                    if association_id:
                        try:
                            ec2_client.disassociate_address(AssociationId=association_id)
                        except Exception as e:
                            errors.append(f"Instance {instance_id}: Lỗi khi gỡ IP cũ - {str(e)}")
                            failed_count += 1
                            continue
                    
                    # Release old Elastic IP if exists
                    if allocation_id:
                        try:
                            ec2_client.release_address(AllocationId=allocation_id)
                        except Exception as e:
                            # Continue even if release fails
                            pass
                    
                    # Allocate new Elastic IP
                    try:
                        allocation = ec2_client.allocate_address(Domain='vpc')
                        new_allocation_id = allocation['AllocationId']
                        new_public_ip = allocation['PublicIp']
                    except Exception as e:
                        errors.append(f"Instance {instance_id}: Lỗi khi cấp phát IP mới - {str(e)}")
                        failed_count += 1
                        continue
                    
                    # Associate new Elastic IP with instance
                    try:
                        ec2_client.associate_address(
                            AllocationId=new_allocation_id,
                            NetworkInterfaceId=network_interface_id
                        )
                        renewed_count += 1
                        
                        # Update output method display if exists
                        if hasattr(self, 'out_m') and self.out_m:
                            # Update the proxy list display
                            self.update_proxy_display(instance_id, new_public_ip)
                            
                    except Exception as e:
                        # If association fails, release the newly allocated IP
                        try:
                            ec2_client.release_address(AllocationId=new_allocation_id)
                        except:
                            pass
                        errors.append(f"Instance {instance_id}: Lỗi khi gán IP mới - {str(e)}")
                        failed_count += 1
                        continue
                    
                except Exception as e:
                    errors.append(f"Instance {instance_id}: {str(e)}")
                    failed_count += 1
                    continue
            
            # Close loading dialog
            loading.close()
            
            # Show results
            result_message = f"Làm mới IP hoàn tất!\n\n"
            result_message += f"Thành công: {renewed_count}\n"
            result_message += f"Thất bại: {failed_count}\n"
            
            if errors:
                result_message += f"\nChi tiết lỗi:\n"
                for error in errors[:5]:  # Show first 5 errors
                    result_message += f"- {error}\n"
                if len(errors) > 5:
                    result_message += f"... và {len(errors) - 5} lỗi khác"
            
            if renewed_count > 0:
                messagebox.showinfo("Thành công", result_message)
            else:
                messagebox.showerror("Lỗi", result_message)
            
            # Refresh proxy list display
            if hasattr(self, 'refresh_proxy_list'):
                self.refresh_proxy_list()
                
        except Exception as e:
            loading.close()
            messagebox.showerror("Lỗi", f"Lỗi khi làm mới IP: {str(e)}")
            
    except Exception as e:
        messagebox.showerror("Lỗi", f"Lỗi không xác định: {str(e)}")

def update_proxy_display(self, instance_id, new_ip):
    """Update proxy display with new IP address"""
    try:
        if hasattr(self, 'out_m') and self.out_m:
            # Get current proxy list text
            current_text = self.out_m.get("1.0", "end-1c")
            
            # Update the IP for this instance in the display
            lines = current_text.split('\n')
            updated_lines = []
            
            for line in lines:
                if instance_id in line:
                    # Parse and update the line with new IP
                    parts = line.split(':')
                    if len(parts) >= 2:
                        # Reconstruct line with new IP
                        parts[0] = new_ip
                        updated_lines.append(':'.join(parts))
                    else:
                        updated_lines.append(line)
                else:
                    updated_lines.append(line)
            
            # Update the text widget
            self.out_m.delete("1.0", "end")
            self.out_m.insert("1.0", '\n'.join(updated_lines))
            
    except Exception as e:
        print(f"Lỗi khi cập nhật hiển thị: {str(e)}")

def refresh_proxy_list(self):
    """Refresh the proxy list display with current instance information"""
    try:
        if not self.ak.get('access_key') or not self.ak.get('secret_key'):
            return
        
        if not hasattr(self, 'instances') or not self.instances:
            return
        
        # Initialize EC2 client
        ec2_client = boto3.client(
            'ec2',
            aws_access_key_id=self.ak['access_key'],
            aws_secret_access_key=self.ak['secret_key'],
            region_name=self.ak.get('region', 'us-east-1')
        )
        
        # Clear current display
        if hasattr(self, 'out_m') and self.out_m:
            self.out_m.delete("1.0", "end")
            
            proxy_lines = []
            
            for instance_id in self.instances:
                try:
                    response = ec2_client.describe_instances(InstanceIds=[instance_id])
                    
                    if response['Reservations']:
                        instance = response['Reservations'][0]['Instances'][0]
                        
                        # Get public IP
                        public_ip = instance.get('PublicIpAddress', 'N/A')
                        
                        # Get proxy port (default or from tags)
                        proxy_port = '3128'  # Default proxy port
                        
                        # Get username and password from tags or use defaults
                        username = 'proxy_user'
                        password = 'proxy_pass'
                        
                        if 'Tags' in instance:
                            for tag in instance['Tags']:
                                if tag['Key'] == 'ProxyPort':
                                    proxy_port = tag['Value']
                                elif tag['Key'] == 'ProxyUser':
                                    username = tag['Value']
                                elif tag['Key'] == 'ProxyPass':
                                    password = tag['Value']
                        
                        # Format: IP:PORT:USER:PASS
                        proxy_line = f"{public_ip}:{proxy_port}:{username}:{password}"
                        proxy_lines.append(proxy_line)
                        
                except Exception as e:
                    print(f"Lỗi khi lấy thông tin instance {instance_id}: {str(e)}")
                    continue
            
            # Update display
            if proxy_lines:
                self.out_m.insert("1.0", '\n'.join(proxy_lines))
                
    except Exception as e:
        print(f"Lỗi khi làm mới danh sách proxy: {str(e)}")
